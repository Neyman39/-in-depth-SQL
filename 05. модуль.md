### Задание А-1. Продажи ниже себестоимости

```sql
-- Задание А-1.1. Количество позиций, где цена со скидкой ниже себестоимости
SELECT COUNT(*) AS items_below_cost
FROM   order_items AS oi
JOIN   orders      AS o  ON o.order_id = oi.order_id
WHERE  o.status <> 'Отменён'
  AND  oi.unit_price * (1 - oi.discount_pct / 100.0)
         < (SELECT p.standard_cost
            FROM   products AS p
            WHERE  p.product_id = oi.product_id);
-- Проверка: 879

-- Задание А-1.2. Группировка по продукту с названием через подзапрос в SELECT
SELECT oi.product_id,
       (SELECT p.product_name
        FROM   products AS p
        WHERE  p.product_id = oi.product_id) AS product_name,
       COUNT(*)                                AS items_cnt,
       SUM(oi.quantity)                        AS total_qty,
       ROUND(SUM(
           oi.quantity * (
               (SELECT p.standard_cost
                FROM   products AS p
                WHERE  p.product_id = oi.product_id)
               - oi.unit_price * (1 - oi.discount_pct / 100.0)
           )
       ), 2)                                   AS loss_rub
FROM   order_items AS oi
JOIN   orders      AS o  ON o.order_id = oi.order_id
WHERE  o.status <> 'Отменён'
  AND  oi.unit_price * (1 - oi.discount_pct / 100.0)
         < (SELECT p.standard_cost
            FROM   products AS p
            WHERE  p.product_id = oi.product_id)
GROUP  BY oi.product_id
ORDER  BY loss_rub DESC;
-- Проверка: 2 продукта
```

```sql
           SELECT 1
           FROM   orders AS o
```

### Задание А-5. Продукты с повышенной долей возвратов

```sql
-- Задание А-5. Продукты с долей возвратов выше средней по компании
SELECT sub.product_id,
       (SELECT p.product_name
        FROM   products AS p
        WHERE  p.product_id = sub.product_id) AS product_name,
       (SELECT p.shelf_life_days
        FROM   products AS p
        WHERE  p.product_id = sub.product_id) AS shelf_life_days,
       sub.returned_qty,
       sub.sold_qty,
       ROUND(100.0 * sub.returned_qty / sub.sold_qty, 2) AS return_pct
FROM (
    SELECT oi.product_id,
           COALESCE((SELECT SUM(r.quantity)
                     FROM   returns AS r
                     WHERE  r.product_id = oi.product_id), 0) AS returned_qty,
           SUM(oi.quantity) AS sold_qty
    FROM   order_items AS oi
    JOIN   orders      AS o ON o.order_id = oi.order_id
    WHERE  o.status <> 'Отменён'
    GROUP  BY oi.product_id
) AS sub
WHERE  100.0 * sub.returned_qty / sub.sold_qty
         > (SELECT 100.0 * SUM(r.quantity) / SUM(oi.quantity)
            FROM   returns    AS r
            JOIN   order_items AS oi ON oi.product_id = r.product_id
            JOIN   orders      AS o  ON o.order_id  = oi.order_id
            WHERE  o.status <> 'Отменён')
ORDER  BY return_pct DESC;
-- Проверка: средняя доля ≈ 0.41%, 6 продуктов

-- Задание А-5 (бонус). Доля возвратов скоропорта (≤10 дней) от клиентов Сибирского ФО
SELECT ROUND(
    100.0 *
    (SELECT SUM(r.quantity)
     FROM   returns    AS r
     JOIN   order_items AS oi ON oi.product_id = r.product_id AND oi.order_id = r.order_id
     JOIN   orders      AS o  ON o.order_id = oi.order_id
     JOIN   customers   AS c  ON c.customer_id = o.customer_id
     WHERE  oi.product_id IN (SELECT product_id FROM products WHERE shelf_life_days <= 10)
       AND  c.region_id IN (SELECT region_id FROM regions WHERE federal_district = 'Сибирский'))
    /
    (SELECT SUM(oi.quantity)
     FROM   order_items AS oi
     JOIN   orders      AS o ON o.order_id = oi.order_id
     JOIN   customers   AS c ON c.customer_id = o.customer_id
     WHERE  o.status <> 'Отменён'
       AND  oi.product_id IN (SELECT product_id FROM products WHERE shelf_life_days <= 10)
       AND  c.region_id IN (SELECT region_id FROM regions WHERE federal_district = 'Сибирский'))
, 2) AS siberia_short_life_return_pct;
```

### Задание А-6. Долг выше кредитного лимита

```sql
-- Задание А-6. Клиенты, у которых текущий долг превышает credit_limit
SELECT c.customer_id,
       c.customer_name,
       c.customer_type,
       c.credit_limit,
       COALESCE((SELECT SUM(i.amount)
                 FROM   invoices AS i
                 JOIN   orders   AS o ON o.order_id = i.order_id

                 JOIN   orders   AS o ON o.order_id = i.order_id

     - COALESCE((SELECT SUM(p.amount)
                 FROM   payments AS p
                 JOIN   invoices AS i ON i.invoice_id = p.invoice_id
                 JOIN   orders   AS o ON o.order_id = i.order_id
                 WHERE  o.customer_id = c.customer_id), 0) AS current_debt,

                 FROM   invoices AS i
                 JOIN   orders   AS o ON o.order_id = i.order_id

                  JOIN   orders   AS o ON o.order_id = i.order_id
```

### Задание А-8 ★. Ловушка NOT IN

```sql
-- Задание А-8а. Регионы без дистрибьюторов через NOT IN (ЛОВУШКА!)
SELECT r.region_id, r.region_name
FROM   regions AS r
WHERE  r.region_id NOT IN (
           SELECT c.region_id
           FROM   customers AS c
           WHERE  c.customer_type = 'Дистрибьютор'
       )
ORDER  BY r.region_id;
-- Проверка: 0 строк (из-за NULL в region_id у клиентов)

-- Задание А-8б. Тот же запрос через NOT EXISTS (правильно)
SELECT r.region_id, r.region_name
FROM   regions AS r
WHERE  NOT EXISTS (
           SELECT 1
           FROM   customers AS c
           WHERE  c.customer_type = 'Дистрибьютор'
             AND  c.region_id = r.region_id
       )
ORDER  BY r.region_id;
-- Проверка: 2 строки

-- Пояснение разницы:
-- NOT IN с подзапросом, возвращающим NULL, даёт UNKNOWN для любой строки → 0 результатов.
-- NOT EXISTS корректно обрабатывает NULL: если у региона NULL, сравнение NULL = NULL → FALSE,
-- и строка проходит, если нет дистрибьюторов с таким region_id.
```

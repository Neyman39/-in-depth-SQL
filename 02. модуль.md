### А4. Сроки отгрузки

```sql
-- Задание А4.1. Сроки отгрузки с классификацией
SELECT o.order_id,
       o.order_number,
       o.order_date,
       o.shipped_date,
       o.shipped_date - o.order_date                   AS lead_days,
       c.region_id,
       CASE
           WHEN o.shipped_date - o.order_date BETWEEN 1 AND 3 THEN '1–3 дня'
           WHEN o.shipped_date - o.order_date BETWEEN 4 AND 6 THEN '4–6 дней'
           ELSE '7+ дней'
       END                                             AS lead_class
FROM   orders      AS o
JOIN   customers   AS c ON c.customer_id = o.customer_id
WHERE  o.shipped_date IS NOT NULL
  AND  o.status <> 'Отменён'
ORDER  BY lead_days DESC
LIMIT  10;

-- Задание А4.2. Отгрузки 5+ дней по регионам Сибири и остальных
SELECT COUNT(*) FILTER (WHERE c.region_id IN (9, 10))       AS siberia_long,
       COUNT(*) FILTER (WHERE c.region_id NOT IN (9, 10) 
                           OR c.region_id IS NULL)          AS others_long,
       COUNT(*) FILTER (WHERE c.region_id IN (9, 10))       AS siberia_total,
       COUNT(*) FILTER (WHERE c.region_id NOT IN (9, 10) 
                           OR c.region_id IS NULL)          AS others_total
FROM   orders    AS o
JOIN   customers AS c ON c.customer_id = o.customer_id
WHERE  o.shipped_date IS NOT NULL
  AND  o.status <> 'Отменён'
  AND  o.shipped_date - o.order_date >= 5;
```

### А5. Просрочка оплаты

```sql
-- Задание А5.1. Просрочка оплат с классификацией
SELECT p.payment_id,
       i.invoice_number,
       i.due_date,
       p.payment_date,
       p.payment_date - i.due_date                   AS overdue_days,
       CASE
           WHEN p.payment_date - i.due_date <= 0  THEN 'в срок'
           WHEN p.payment_date - i.due_date <= 30 THEN 'до 30 дней'
           ELSE 'более 30 дней'
       END                                           AS overdue_class,
       o.customer_id
FROM   payments  AS p
JOIN   invoices  AS i ON i.invoice_id = p.invoice_id
JOIN   orders    AS o ON o.order_id   = i.order_id
ORDER  BY overdue_days DESC
LIMIT  10;

-- Шаг 1: Считаем оплаты с просрочкой > 30 дней
SELECT COUNT(*) AS over_30_days
FROM   payments  AS p
JOIN   invoices  AS i ON i.invoice_id = p.invoice_id
WHERE  p.payment_date - i.due_date > 30;
-- 231

-- Шаг 2: Находим клиента с максимальной просрочкой
SELECT o.customer_id,
       MAX(p.payment_date - i.due_date) AS max_overdue
FROM   payments  AS p
JOIN   invoices  AS i ON i.invoice_id = p.invoice_id
JOIN   orders    AS o ON o.order_id   = i.order_id
GROUP  BY o.customer_id
ORDER  BY max_overdue DESC
LIMIT  1;

SELECT COUNT(*) AS top_customer_over_30

JOIN   orders    AS o ON o.order_id   = i.order_id
WHERE  o.customer_id = 1  -- ← подставьте customer_id из шага 2
  AND  p.payment_date - i.due_date > 30;
```

### А6. Цена со скидкой против себестоимости

```sql
-- Задание А6.1. Позиции с отрицательной маржой с 01.03.2025
SELECT oi.order_item_id,
       o.order_date,
       p.product_name,
       oi.unit_price,
       oi.discount_pct,
       ROUND(oi.unit_price * (1 - oi.discount_pct / 100.0), 2)                       AS net_price,
       p.standard_cost,
       ROUND(oi.unit_price * (1 - oi.discount_pct / 100.0) - p.standard_cost, 2)     AS margin_unit,
       ROUND(100.0 * (oi.unit_price * (1 - oi.discount_pct / 100.0) - p.standard_cost)
                   / p.standard_cost, 1)                                               AS margin_pct
FROM   order_items AS oi
JOIN   orders      AS o ON o.order_id   = oi.order_id
JOIN   products    AS p ON p.product_id = oi.product_id
WHERE  oi.unit_price * (1 - oi.discount_pct / 100.0) < p.standard_cost
  AND  o.order_date >= DATE '2025-03-01'
  AND  o.status <> 'Отменён'
ORDER  BY margin_unit ASC
LIMIT  10;

-- Задание А6.2 (★). Сумма потерь
SELECT COUNT(*) AS loss_items,
       ROUND(SUM(oi.quantity * (p.standard_cost - oi.unit_price * (1 - oi.discount_pct / 100.0))), 2) AS total_loss

JOIN   orders      AS o ON o.order_id   = oi.order_id
JOIN   products    AS p ON p.product_id = oi.product_id

  AND  o.order_date >= DATE '2025-03-01'
```

```sql
-- Задание А2.1. Долг клиентов с сравнением с лимитом
SELECT c.customer_id,
       c.customer_name,
       c.credit_limit,
       COUNT(DISTINCT i.invoice_id)                            AS invoices_cnt,
       ROUND(SUM(i.amount), 2)                                 AS invoiced,
       ROUND(SUM(COALESCE(pay.paid, 0)), 2)                    AS paid,
       ROUND(SUM(i.amount - COALESCE(pay.paid, 0)), 2)         AS debt,
       ROUND(SUM(i.amount - COALESCE(pay.paid, 0))
             FILTER (WHERE i.due_date < DATE '2026-01-01'), 2) AS overdue_debt,
       ROUND(SUM(i.amount - COALESCE(pay.paid, 0)) / c.credit_limit, 2) AS debt_to_limit
FROM   customers AS c
JOIN   orders    AS o ON o.customer_id = c.customer_id
JOIN   invoices  AS i ON i.order_id    = o.order_id
LEFT   JOIN (SELECT invoice_id, SUM(amount) AS paid
             FROM   payments
             GROUP  BY invoice_id) AS pay ON pay.invoice_id = i.invoice_id
GROUP  BY c.customer_id, c.customer_name, c.credit_limit
HAVING SUM(i.amount - COALESCE(pay.paid, 0)) > c.credit_limit
ORDER  BY debt DESC;
```

### А4. Выручка по кварталам с ROLLUP

```sql
-- Задание А4. Отчёт «год / квартал / выручка» с подытогами
SELECT CASE WHEN GROUPING(t.yr)  = 1 THEN 'ВСЕГО'
            ELSE t.yr::text END                                      AS yr,
       CASE WHEN GROUPING(t.yr)  = 1 THEN ''
            WHEN GROUPING(t.qtr) = 1 THEN 'Итого за год'
            ELSE 'Q' || t.qtr END                                    AS qtr,
       ROUND(SUM(t.revenue) / 1e6, 2)                                AS revenue_mln
FROM  (SELECT EXTRACT(YEAR    FROM o.order_date)::int AS yr,
              EXTRACT(QUARTER FROM o.order_date)::int AS qtr,
              oi.quantity * oi.unit_price * (1 - oi.discount_pct / 100) AS revenue
       FROM   orders      AS o
       JOIN   order_items AS oi ON oi.order_id = o.order_id
       WHERE  o.status <> 'Отменён') AS t
GROUP  BY ROLLUP (t.yr, t.qtr)
ORDER  BY t.yr NULLS LAST, t.qtr NULLS LAST;
```

### А6. Маркетинг vs выручка по кварталам

```sql
-- Задание А6. Расходы на маркетинг и выручка по кварталам
WITH revenue AS (
    SELECT EXTRACT(YEAR FROM o.order_date)::int            AS yr,
           EXTRACT(QUARTER FROM o.order_date)::int         AS qtr,
           SUM(oi.quantity * oi.unit_price * (1 - oi.discount_pct / 100)) AS revenue
    FROM   orders      AS o
    JOIN   order_items AS oi ON oi.order_id = o.order_id
    WHERE  o.status <> 'Отменён'
    GROUP  BY EXTRACT(YEAR FROM o.order_date), EXTRACT(QUARTER FROM o.order_date)
),
marketing AS (
    SELECT EXTRACT(YEAR FROM expense_date)::int            AS yr,
           EXTRACT(QUARTER FROM expense_date)::int         AS qtr,
           SUM(amount)                                     AS mkt_expense
    FROM   expenses
    WHERE  expense_category = 'Маркетинг'
    GROUP  BY EXTRACT(YEAR FROM expense_date), EXTRACT(QUARTER FROM expense_date)
)
SELECT r.yr,
       'Q' || r.qtr                                        AS qtr,
       ROUND(r.revenue / 1e6, 2)                           AS revenue_mln,
       ROUND(COALESCE(m.mkt_expense, 0) / 1e6, 2)          AS mkt_mln,
       ROUND(100.0 * COALESCE(m.mkt_expense, 0) / r.revenue, 2) AS mkt_share_pct
FROM   revenue r
LEFT   JOIN marketing m ON m.yr = r.yr AND m.qtr = r.qtr
ORDER  BY r.yr, r.qtr;
```

### А7. Дубли клиентов

```sql
-- Задание А7. ИНН, которым соответствует более одной карточки клиента
SELECT inn,
       COUNT(*)                                                          AS cards,
       STRING_AGG(customer_name, ' | ' ORDER BY customer_id)             AS names,
       ARRAY_AGG(customer_id ORDER BY customer_id)                       AS ids,
       COUNT(DISTINCT manager_id)                                        AS managers_cnt
FROM   customers
GROUP  BY inn
HAVING COUNT(*) > 1
ORDER  BY cards DESC;
```
